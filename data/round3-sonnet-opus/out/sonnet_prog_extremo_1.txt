```python
def sorpresa(nums):
    pares = sorted(n for n in nums if n % 2 == 0)
    impares = sorted((n for n in nums if n % 2), reverse=True)
    return {"suma": sum(nums), "pares": pares, "impares": impares,
            "mayor_diferencia": max(nums) - min(nums),
            "acumulada": [sum(nums[:i + 1]) for i in range(len(nums))]}

print(sorpresa([7, 2, 9, 4, 5, 8]))
```
