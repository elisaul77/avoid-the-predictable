def collatz_viaje(nums):
    """Para cada número, cuenta los pasos de Collatz hasta llegar a 1."""
    def pasos(n):
        c = 0
        while n != 1:
            n = n // 2 if n % 2 == 0 else 3 * n + 1
            c += 1
        return c
    return {n: pasos(n) for n in nums if n > 0}
